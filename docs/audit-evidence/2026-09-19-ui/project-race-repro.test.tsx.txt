import { act, cleanup, fireEvent, render, screen, waitFor } from "@testing-library/react";
import { afterEach, expect, it, vi } from "vitest";
import { withWorkspaceClient } from "@deckastra/workspace-client/testing";
import { DeckList } from "../src/components/DeckList";
vi.mock("@deckastra/renderer/react", () => ({ ScaledSlide: () => <div /> }));
afterEach(() => { cleanup(); vi.unstubAllGlobals(); });
it("audit: a late response from project A must not replace project B's cards", async () => {
  let release!: (value: Response) => void;
  let started = false;
  const held = new Promise<Response>(resolve => { release = resolve; });
  const reply = (body: unknown) => new Response(JSON.stringify(body));
  vi.stubGlobal("fetch", vi.fn(async (url: string) => {
    const path = String(url);
    if (path.includes("/account")) return reply({ user: { id: "usr_test" }, capabilities: { sharing: false, checkpoints: false }, workspaces: [{ id: "wsp_test", name: "Workspace", role: "owner", projects: [{ id: "prj_a", name: "Project A" }, { id: "prj_b", name: "Project B" }] }] });
    if (path.includes("/projects/prj_a/presentations")) { started = true; return held; }
    if (path.includes("/projects/prj_b/presentations")) return reply({ presentations: [{ id: "doc_b", title: "B only deck", version_id: null, slide_count: 1 }] });
    return new Response("{}", { status: 404 });
  }));
  render(<DeckList onOpen={() => {}} />, { wrapper: withWorkspaceClient() });
  await waitFor(() => expect(started).toBe(true));
  fireEvent.click(screen.getByRole("button", { name: /Project B/ }));
  await screen.findByText("B only deck");
  await act(async () => release(reply({ presentations: [{ id: "doc_a", title: "A only deck", version_id: null, slide_count: 1 }] })));
  expect(screen.queryByText("A only deck")).toBeNull();
  expect(screen.queryByText("B only deck")).not.toBeNull();
});
